# Csound headers
#
# csound.h includes "version.h" and csound_types.h may include
# "float-version.h", but the csound submodule only ships the *.in templates:
# the real headers are generated by csound's own CMake, which is not run here.
# Generate them from the templates in the build directory instead of borrowing
# the pre-generated copies from the Android jni tree, so a plain submodule
# checkout (or an installed Csound header set) is not required.
set(CSOUND_GENERATED_INCLUDE_DIR ${CMAKE_CURRENT_BINARY_DIR}/csound-include)
file(MAKE_DIRECTORY ${CSOUND_GENERATED_INCLUDE_DIR})

# Major version of the headers being built against. Csound validates it in
# csoundModuleInfo() and refuses plugins whose major version differs.
# Minor/patch are reported as 0: Csound only rejects plugins that claim to be
# *newer* than the host, so 0 keeps the binaries usable on any 7.x Csound.
file(STRINGS "${CMAKE_SOURCE_DIR}/csound/CMakeLists.txt" _csound_version_line
     REGEX "^[ \t]*set\\(CS_VERSION[ \t]+\"?[0-9]+\"?[ \t]*\\)")
if(_csound_version_line MATCHES "set\\(CS_VERSION[ \t]+\"?([0-9]+)")
    set(CS_VERSION ${CMAKE_MATCH_1})
else()
    set(CS_VERSION 7)
endif()
set(CS_SUBVER 0)
set(CS_PATCHLEVEL 0)
set(CS_PACKAGE_DATE_DEFINE "__DATE__")

configure_file("${CMAKE_SOURCE_DIR}/csound/include/version.h.in"
               "${CSOUND_GENERATED_INCLUDE_DIR}/version.h")
configure_file("${CMAKE_SOURCE_DIR}/csound/include/float-version.h.in"
               "${CSOUND_GENERATED_INCLUDE_DIR}/float-version.h")

# Include path used to build every plugin, shared with the API probe below.
# The generated directory comes first so the templates always win over a
# stale header that may be left in csound/include by a manual csound build.
set(CSOUND_INCLUDE_DIRS ${CSOUND_GENERATED_INCLUDE_DIR} ${CMAKE_SOURCE_DIR}/csound/include)
if(LINUX)
    list(APPEND CSOUND_INCLUDE_DIRS /usr/local/include/csound)
endif()
# Directory-wide include so that subprojects (jsfx/jsusfx) resolve version.h
# from the generated directory instead of relying on a copy in the csound tree.
include_directories(${CSOUND_GENERATED_INCLUDE_DIR})

# Csound 6 is no longer supported. Verify that the headers we are about to
# build against really are Csound 7: the snippet below uses the OPDS layout of
# the Csound 7 API (init/perf), which Csound 6 lacks (it has iopadr/opadr).
# Because it is compiled with the same include path as the plugins, it also
# catches a csound submodule left on the 'csound6' branch.
include(CheckCSourceCompiles)
# Do not cache the result: probe on every configure so that switching the
# csound submodule branch is detected without wiping the build directory.
unset(CSOUND_HEADERS_ARE_V7 CACHE)
set(CMAKE_REQUIRED_INCLUDES ${CSOUND_INCLUDE_DIRS})
check_c_source_compiles("
#include \"csdl.h\"
int main(void) {
    OPDS *opds = 0;
    void *init = (void *) opds->init;
    void *perf = (void *) opds->perf;
    (void) init;
    (void) perf;
    return 0;
}" CSOUND_HEADERS_ARE_V7)
unset(CMAKE_REQUIRED_INCLUDES)

if(NOT CSOUND_HEADERS_ARE_V7)
    message(FATAL_ERROR
        "The Csound headers found in:\n"
        "  ${CSOUND_INCLUDE_DIRS}\n"
        "do not provide the Csound 7 API (OPDS.init/OPDS.perf). "
        "Csound 6 is no longer supported.\n"
        "Check that the 'csound' submodule is on the 'develop' branch "
        "(git submodule update --init csound).")
endif()
message(STATUS "Csound headers: Csound 7 API confirmed (version ${CS_VERSION})")

function(addflag flag flagname)
    check_c_compiler_flag(${flag} ${flagname})
    if (${flagname})
        set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${flag}" PARENT_SCOPE)
    endif()
    check_cxx_compiler_flag(${flag} CXX_${flagname})
    if (CXX_${flagname})
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${flag}" PARENT_SCOPE)
    endif()
endfunction(addflag)


MACRO(SUBDIRLIST result curdir)
    FILE(GLOB children RELATIVE ${curdir} ${curdir}/*)
    SET(dirlist "")
    FOREACH(child ${children})
        message(STATUS "looking at ${child}")
        IF(IS_DIRECTORY ${curdir}/${child})
            LIST(APPEND dirlist ${child})
        ENDIF()
    ENDFOREACH()
    SET(${result} ${dirlist})
ENDMACRO()


MACRO(ADD_ALL_SUBDIRECTORIES directory)
    subdirlist(SUBDIRS ${directory})
    message(STATUS "Found subdirs: ${SUBDIRS}")
    foreach(SUBDIR ${SUBDIRS})
        set(thissubdir "${directory}/${SUBDIR}")
        if(EXISTS "${directory}/${SUBDIR}/CMakeLists.txt")
            message(STATUS "Addind subdir: ${thissubdir}")
            add_subdirectory(${directory}/${SUBDIR})
        else()
            message(WARNING "Skipping ${directory}/${SUBDIR} because no CMakeLists.txt file was found")
        endif()
    endforeach()
ENDMACRO()

function(add_shared_library target srcs)
    if(APPLE)
        add_library(${pluginlib} SHARED ${srcs})
    else()
        add_library(${pluginlib} MODULE ${srcs})
    endif()
endfunction(add_shared_library)

function(make_c_plugin pluginlib srcs)
    message(STATUS "make_c_plugin: ${pluginlib}")
    add_shared_library(${pluginlib} ${srcs})
    # csound/Android.../jni is included because version.h exists only as
    # version.h.in in csound/include and needs an extra cmake step to generate it
    target_include_directories(${pluginlib} PRIVATE ${CSOUND_INCLUDE_DIRS})
    target_include_directories(${pluginlib} PRIVATE ${CMAKE_SOURCE_DIR}/src/common)

    # target_compile_definitions(${pluginlib} PRIVATE USE_DOUBLE)
    set_property(TARGET ${pluginlib} PROPERTY C_STANDARD 99)
    set_target_properties(${pluginlib} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${BUILD_PLUGINS_DIR}
        LIBRARY_OUTPUT_DIRECTORY ${BUILD_PLUGINS_DIR}
    )

    install(TARGETS ${pluginlib} LIBRARY DESTINATION "${PLUGIN_INSTALL_DIR}" )
endfunction(make_c_plugin)

# pluginlib: name of the plugin
# srcs: sources of the plugin
function(make_plugin pluginlib srcs)
    message(STATUS "make_plugin: ${pluginlib}, sources: ${srcs}")
    add_shared_library(${pluginlib} ${srcs})
    target_include_directories(${pluginlib} PRIVATE ${CSOUND_INCLUDE_DIRS})

    # target_compile_definitions(${pluginlib} PRIVATE USE_DOUBLE)
    set_target_properties(${pluginlib} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${BUILD_PLUGINS_DIR}
        LIBRARY_OUTPUT_DIRECTORY ${BUILD_PLUGINS_DIR}
    )
    install(TARGETS ${pluginlib} LIBRARY DESTINATION "${PLUGIN_INSTALL_DIR}")

endfunction(make_plugin)
